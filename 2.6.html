Panduan Penggunaan

  1. Unggah sketch program Arduino ke modul Arduino Uno Anda.
  2. Hubungkan Push Button 1 ke Pin 2 dan Push Button 2 ke Pin 3 (menggunakan konfigurasi INPUT_PULLUP).
  3. Hubungkan LED Merah ke Pin 8 dan LED Biru ke Pin 9.
  4. Gunakan browser Chrome atau Edge, lalu klik tombol "Hubungkan Arduino" di kanan atas.
  5. Pilih port USB COM Arduino Anda pada jendela pop-up untuk mulai memantau simulasi secara real-time.

dua_button_dua_led.ino 
// ======================================================
// PROGRAM KONTROL 2 LED DENGAN 2 PUSH BUTTON
// Pin Digital 2  : Push Button 1 (Input Pull-Up)
// Pin Digital 3  : Push Button 2 (Input Pull-Up)
// Pin Digital 8  : LED Merah (Output)
// Pin Digital 9  : LED Biru (Output)
// Baud Rate      : 9600 bps
// ======================================================

const int BUTTON1_PIN = 2;
const int BUTTON2_PIN = 3;
const int RED_PIN     = 8;
const int BLUE_PIN    = 9;

void setup() {
  Serial.begin(9600);

  pinMode(BUTTON1_PIN, INPUT_PULLUP);
  pinMode(BUTTON2_PIN, INPUT_PULLUP);
  pinMode(RED_PIN, OUTPUT);
  pinMode(BLUE_PIN, OUTPUT);
}

void loop() {
  int btn1State = digitalRead(BUTTON1_PIN);
  int btn2State = digitalRead(BUTTON2_PIN);

  // Logika Tombol 1 Ditekan (Active LOW)
  if (btn1State == LOW) {
    digitalWrite(RED_PIN, HIGH);
    digitalWrite(BLUE_PIN, LOW);
    Serial.println("BTN1_PRESSED");
  } 
  // Logika Tombol 2 Ditekan (Active LOW)
  else if (btn2State == LOW) {
    digitalWrite(RED_PIN, LOW);
    digitalWrite(BLUE_PIN, HIGH);
    Serial.println("BTN2_PRESSED");
  } 
  // Kondisi default saat kedua tombol dilepas
  else {
    digitalWrite(RED_PIN, LOW);
    digitalWrite(BLUE_PIN, LOW);
    Serial.println("BOTH_RELEASED");
  }

  delay(100);
}
