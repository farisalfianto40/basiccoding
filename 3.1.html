// Pin A0: analogRead | Pin 3: analogWrite (PWM)
int pinSuhu = A0;
int pinAC = 3;
 
void setup() {
Serial.begin(9600);
pinMode(pinAC, OUTPUT);
}
 
void loop() {
int adcVal = analogRead(pinSuhu);
int pwmVal = map(adcVal, 0, 1023, 0, 255);
analogWrite(pinAC, pwmVal);
Serial.print("[analogRead A0]: "); Serial.print(adcVal);
Serial.print(" | [analogWrite PWM]: "); Serial.println(pwmVal);
delay(300);
}
