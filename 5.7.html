Studi Kasus: Boiler Steam Relief & Manhole Interlock

Industri Kelapa Sawit (CPO) & Tekstil
High-Pressure Steam Boiler Safety System

Pengamanan Tangki Uap Bertekanan Kritis & Interlock Pintu Manhole

Sistem proteksi otomatis untuk mencegah ledakan boiler uap pada pabrik kelapa sawit (sterilizer CPO) dan mesin katun tekstil. Mengukur suhu air via Sensor DS18B20 dan simulator tekanan via Potensio 10K. Saat batas kritis terlampaui (>110°C / >6.0 Bar), Solenoid Door Lock langsung mengunci pintu manhole secara fisik, Safety Relief Valve (Servo) berputar membuang uap berlebih, dan sirine darurat meraung.

Panduan Penggunaan Simulasi Interlock Safety Boiler

  1. Rangkai komponen modul lab: Sensor Suhu DS18B20 ke Pin D2 (gunakan resistor pull-up 4.7kΩ), Potensio 10K (Simulator Tekanan Bar) ke Pin Analog A0, Servo Motor Safety Relief Valve ke Pin D9, Solenoid Door Lock 12V / Relay Kunci Manhole ke Pin D5, dan Buzzer Alarm ke Pin D6.
  2. Unggah program C++ ke papan Arduino Uno (pastikan pustaka OneWire, DallasTemperature, dan Servo telah terpasang).
  3. Klik tombol "Hubungkan Arduino" dan pilih port serial USB yang aktif.
  4. Kondisi Normal (Suhu ≤ 110.0°C DAN Tekanan ≤ 6.0 Bar): Katup buang Servo berada di posisi 0° (TERTUTUP), Solenoid Door Lock UNLOCKED (pintu manhole dapat dibuka aman saat perawatan), dan Buzzer hening.
  5. Uji Kondisi Kritis Overpressure / Overtemp (> 110.0°C atau > 6.0 Bar): Putar Potensio 10K hingga simulasi tekanan melampaui 6.0 Bar atau panaskan probe DS18B20. Sistem seketika mengaktifkan Solenoid Door Lock (LOCKED) agar pintu manhole terkunci rapat demi mencegah bahaya operator tersiram uap, Servo berputar 90° membuang uap (Safety Relief Valve BUKA), serta Buzzer meraung.

// =========================================================================
// PROGRAM BOILER STEAM RELIEF & MANHOLE INTERLOCK (CPO & TEKSTIL)
// JURUSAN OTOMASI INDUSTRI / TEKNIK KIMIA - SMK SMTI YOGYAKARTA
// -------------------------------------------------------------------------
// Pin Digital 2 : DS18B20 Sensor Suhu Air Boiler (OneWire Bus)
// Pin Analog A0 : Potensio 10K (Simulator Tekanan Uap Bar 0.0 - 10.0 Bar)
// Pin Digital 5 : Solenoid Door Lock 12V / Relay Kunci Manhole (HIGH = Locked)
// Pin Digital 6 : Buzzer Sirine Alarm Bahaya
// Pin Digital 9 : Servo Motor (Safety Relief Valve - 0 deg = CLOSED, 90 deg = OPEN)
// Baud Rate     : 9600 bps
// =========================================================================

#include <OneWire.h>
#include <DallasTemperature.h>
#include <Servo.h>

#define ONE_WIRE_BUS 2
#define PRESS_POT_PIN A0
#define SOLENOID_LOCK_PIN 5
#define BUZZER_PIN 6
#define SERVO_RELIEF_PIN 9

// Batas Kritis Keselamatan Boiler
const float CRITICAL_TEMP = 110.0; // Batas Kritis Suhu (°C)
const float CRITICAL_PRESS = 6.0;  // Batas Kritis Tekanan (Bar)

OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensors(&oneWire);
Servo reliefValve;

void setup() {
  Serial.begin(9600);

  sensors.begin();
  reliefValve.attach(SERVO_RELIEF_PIN);
  reliefValve.write(0); // Inisialisasi Katup Tutup (0 Derajat)

  pinMode(PRESS_POT_PIN, INPUT);
  pinMode(SOLENOID_LOCK_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  digitalWrite(SOLENOID_LOCK_PIN, LOW); // Door Lock Unlocked
  digitalWrite(BUZZER_PIN, LOW);        // Buzzer Mute
}

void loop() {
  // 1. Baca Suhu Air Boiler dari Sensor DS18B20
  sensors.requestTemperatures();
  float tempC = sensors.getTempCByIndex(0);
  if (tempC == DEVICE_DISCONNECTED_C) {
    tempC = 25.0; // Fallback jika probe terlepas
  }

  // 2. Baca Simulasi Tekanan Uap dari Potensio 10K (0 - 10.0 Bar)
  int rawADC = analogRead(PRESS_POT_PIN);
  float pressureBar = (rawADC / 1023.0) * 10.0;

  // 3. Evaluasi Kondisi Interlock Kritis
  bool isEmergency = (tempC > CRITICAL_TEMP || pressureBar > CRITICAL_PRESS);
  String statusStr = "NORMAL";

  if (isEmergency) {
    statusStr = "CRITICAL_RELIEF";
    
    // Eksekusi Tindakan Pengamanan Darurat:
    // A. Solenoid mengunci pintu manhole secara fisik
    digitalWrite(SOLENOID_LOCK_PIN, HIGH);
    // B. Servo berputar 90 derajat membuka Katup Relief Buang Uap
    reliefValve.write(90);
    // C. Sirine alarm meraung
    digitalWrite(BUZZER_PIN, HIGH);
  } else {
    statusStr = "NORMAL";
    
    // Kondisi Aman Normal:
    digitalWrite(SOLENOID_LOCK_PIN, LOW);
    reliefValve.write(0);
    digitalWrite(BUZZER_PIN, LOW);
  }

  // 4. Kirim Telemetry Terformat via Serial
  Serial.print("TEMP:");
  Serial.print(tempC, 1);
  Serial.print("|PRESS:");
  Serial.print(pressureBar, 1);
  Serial.print("|SERVO:");
  Serial.print(isEmergency ? 90 : 0);
  Serial.print("|LOCK:");
  Serial.print(isEmergency ? 1 : 0);
  Serial.print("|BUZZER:");
  Serial.print(isEmergency ? 1 : 0);
  Serial.print("|STATUS:");
  Serial.println(statusStr);

  delay(500);
}
