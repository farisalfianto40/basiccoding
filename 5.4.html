Panduan Penggunaan Simulasi Interlock

  1. Rangkai komponen modul lab: Sensor Level Air Merah (Resistive) ke Pin Analog A0, Limit Switch Manhole (Pin D4 dengan internal pull-up), Relay Pompa Transfer 5V (Pin D5), dan Buzzer Alarm (Pin D6).
  2. Unggah program Arduino C++ ke papan Arduino Uno.
  3. Klik tombol "Hubungkan Arduino" dan pilih port serial USB yang aktif.
  4. Uji Kondisi Normal (15% – 85%): Jika penutup Manhole tertutup rapat (Limit Switch LOW) dan kedalaman cairan normal, Relay mengaktifkan Pompa Transfer (ON) dan Buzzer hening.
  5. Uji Proteksi Dry-Run (<15%): Angkat sensor dari air atau kura-kurangi level cairan hingga level <15% (nilai ADC sangat rendah). Relay seketika mematikan pompa agar tidak berputar kering (dry-run) dan sirine alarm berbunyi.
  6. Uji Proteksi Overfill (>85%): Celupkan sensor lebih dalam melampaui 85%. Relay mematikan suplai pompa untuk mencegah asam meluap tumpah.
  7. Uji Interlock Manhole Open: Buka saklar Limit Switch manhole. Pompa langsung dimatikan secara otomatis demi keselamatan operator dari uap asam berbahaya.

// =========================================================================
// PROGRAM MONITORING TANGKI MULTI-LEVEL & PROTEKSI DRY-RUN POMPA (PICKLE LINE)
// MENGGUNAKAN SENSOR LEVEL AIR MODUL MERAH (RESISTIVE ANALOG)
// JURUSAN TEKNIK KIMIA & MEKATRONIKA INDUSTRI - SMK SMTI YOGYAKARTA
// -------------------------------------------------------------------------
// Pin Analog A0 : Signal Output Sensor Level Air Merah
// Pin Digital 4 : Limit Switch Manhole Cover (INPUT_PULLUP - LOW = Locked)
// Pin Digital 5 : Relay Pompa Transfer Asam (HIGH = ON, LOW = OFF)
// Pin Digital 6 : Buzzer Alarm Bahaya
// Baud Rate     : 9600 bps
// =========================================================================

#define WATER_SENSOR_PIN A0
#define LIMIT_SW_PIN 4
#define PUMP_RELAY_PIN 5
#define BUZZER_PIN 6

// Nilai Kalibrasi ADC Sensor Level Air Merah
// ADC = 0 saat kering / tidak tersentuh air
// ADC ~ 600-650 saat modul terendam maksimal
const int ADC_MIN = 0;
const int ADC_MAX = 600;

void setup() {
  Serial.begin(9600);

  pinMode(WATER_SENSOR_PIN, INPUT);
  pinMode(LIMIT_SW_PIN, INPUT_PULLUP);
  pinMode(PUMP_RELAY_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  // Inisialisasi awal
  digitalWrite(PUMP_RELAY_PIN, LOW);
  digitalWrite(BUZZER_PIN, LOW);
}

void loop() {
  // 1. Baca Nilai Analog dari Sensor Level Air Merah (0 - 1023)
  int rawADC = analogRead(WATER_SENSOR_PIN);

  // 2. Kalkulasi Persentase Ketinggian Cairan (0.0% - 100.0%)
  float levelPercent = map(rawADC, ADC_MIN, ADC_MAX, 0, 100);
  levelPercent = constrain(levelPercent, 0.0, 100.0);

  // 3. Baca Status Switch Tutup Manhole Tangki (LOW = Terkunci, HIGH = Terbuka)
  bool isCoverClosed = (digitalRead(LIMIT_SW_PIN) == LOW);

  // 4. Logic Interlock Proteksi Pompa
  String statusStr = "NORMAL";
  bool pumpOn = false;
  bool buzzerOn = false;

  if (!isCoverClosed) {
    // BAHAYA 1: Tutup Manhole Terbuka saat Beroperasi
    statusStr = "MANHOLE_OPEN";
    pumpOn = false;
    buzzerOn = true;
  } else if (levelPercent < 15.0) {
    // BAHAYA 2: Level Cairan Terlalu Rendah (Proteksi Dry-Running)
    statusStr = "DRY_RUN_HAZARD";
    pumpOn = false;
    buzzerOn = true;
  } else if (levelPercent > 85.0) {
    // BAHAYA 3: Tangki Hampir Penuh Meluap (Proteksi Overfill)
    statusStr = "OVERFLOW_HAZARD";
    pumpOn = false;
    buzzerOn = true;
  } else {
    // KONDISI NORMAL: Level Antara 15% - 85% & Manhole Terkunci
    statusStr = "NORMAL";
    pumpOn = true;
    buzzerOn = false;
  }

  // Eksekusi Aktuator Relay & Buzzer
  digitalWrite(PUMP_RELAY_PIN, pumpOn ? HIGH : LOW);
  digitalWrite(BUZZER_PIN, buzzerOn ? HIGH : LOW);

  // 5. Kirim Telemetry Terformat via Serial
  Serial.print("LEVEL:");
  Serial.print(levelPercent, 1);
  Serial.print("|ADC:");
  Serial.print(rawADC);
  Serial.print("|LIMIT:");
  Serial.print(isCoverClosed ? 1 : 0);
  Serial.print("|PUMP:");
  Serial.print(pumpOn ? 1 : 0);
  Serial.print("|BUZZER:");
  Serial.print(buzzerOn ? 1 : 0);
  Serial.print("|STATUS:");
  Serial.println(statusStr);

  delay(500);
}
