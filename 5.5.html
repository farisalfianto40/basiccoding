Otomatisasi Pengisian & Penutupan Botol Minuman Kemasan

Sistem mengatur pengisian cairan presisi menggunakan Potensio 10K (Set-point Target Volume) dan Sensor Level Air Merah (A0). Saat volume cairan botol mencapai target, Relay mematikan pompa pengisi, Servo Motor memasang tutup botol (capping 90°), dan Buzzer berbunyi tanda botol selesai dikemas.

Skenario Operasi Pengisian & Penutupan Botol Otomatis

  1. Rangkai modul sesuai pinout: Sensor Level Air Merah ke Pin A0, Potensio 10K ke Pin A1, Servo Motor ke Pin D3, Relay Pompa ke Pin D5, dan Buzzer ke Pin D6.
  2. Putar Potensio 10K untuk menentukan batas target volume isi botol (misal: 300 mL).
  3. Saat botol kosong terdeteksi di stasiun, Relay menyalakan Pompa Pengisi (ON) untuk mengucurkan minuman ke dalam botol.
  4. Sensor level air analog membaca pertambahan ketinggian cairan secara real-time.
  5. Seketika level cairan menyamai/melebihi target volume potensio, Relay mematikan pompa pengisi secara presisi.
  6. Servo Motor Capper berputar 90° bergerak turun menekan/memasang tutup botol, lalu kembali ke posisi awal (0°).
  7. Buzzer berbunyi singkat sebagai sinyal bahwa 1 botol berhasil dikemas dan counter jumlah total botol bertambah (+1).

automatic_bottling_capper.ino 
// =========================================================================
// PROGRAM AUTOMATIC BOTTLING & CAPPING LINE (INDUSTRI F&B)
// MENGGUNAKAN SENSOR LEVEL AIR MERAH (A0) & POTENSIOMETER SET-POINT (A1)
// JURUSAN TEKNIK KIMIA & OTOMASI INDUSTRI - SMK SMTI YOGYAKARTA
// -------------------------------------------------------------------------
// Pin Analog A0 : Sensor Level Air Merah (Actual Liquid Level)
// Pin Analog A1 : Potensio 10K (Set-point Target Volume Botol)
// Pin Digital 3 : Servo Motor (Katup Penutup / Capper Mechanism)
// Pin Digital 5 : Relay Pompa Pengisi Minuman (HIGH = ON, LOW = OFF)
// Pin Digital 6 : Buzzer Signal Batch Complete
// Baud Rate     : 9600 bps
// =========================================================================

#include <Servo.h>

#define LEVEL_SENSOR_PIN A0
#define POT_TARGET_PIN A1
#define SERVO_PIN 3
#define PUMP_RELAY_PIN 5
#define BUZZER_PIN 6

Servo capperServo;

// Kalibrasi Batas Pembacaan Sensor & Potensio (dalam mL)
const int MIN_VOL = 50;   // 50 mL
const int MAX_VOL = 500;  // 500 mL

// Sensor Level Analog ADC Mapping Max (~600)
const int SENSOR_ADC_MAX = 600;

int bottleCount = 0;
enum ProcessState { STATE_FILLING, STATE_CAPPING, STATE_COMPLETED };
ProcessState currentState = STATE_FILLING;

unsigned long stateTimer = 0;

void setup() {
  Serial.begin(9600);

  pinMode(LEVEL_SENSOR_PIN, INPUT);
  pinMode(POT_TARGET_PIN, INPUT);
  pinMode(PUMP_RELAY_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  capperServo.attach(SERVO_PIN);
  capperServo.write(0); // Posisi Servo Awal (Up / Uncapped)

  digitalWrite(PUMP_RELAY_PIN, LOW);
  digitalWrite(BUZZER_PIN, LOW);
}

void loop() {
  // 1. Baca Potensio Set-point Volume Target (50mL - 500mL)
  int potRaw = analogRead(POT_TARGET_PIN);
  int targetVolume = map(potRaw, 0, 1023, MIN_VOL, MAX_VOL);

  // 2. Baca Sensor Level Air Merah (Actual Liquid Level)
  int levelRaw = analogRead(LEVEL_SENSOR_PIN);
  int actualVolume = map(levelRaw, 0, SENSOR_ADC_MAX, 0, MAX_VOL);
  actualVolume = constrain(actualVolume, 0, MAX_VOL);

  bool pumpState = false;
  int servoPos = 0;
  bool buzzerState = false;
  String statusStr = "FILLING";

  // 3. FSM (Finite State Machine) Lini Otomasi Botol
  switch (currentState) {
    case STATE_FILLING:
      statusStr = "FILLING";
      pumpState = true;
      servoPos = 0;
      buzzerState = false;

      // Jika Ketinggian Cairan Mencapai Target Volume
      if (actualVolume >= targetVolume) {
        digitalWrite(PUMP_RELAY_PIN, LOW); // Matikan Pompa Presisi
        currentState = STATE_CAPPING;
        stateTimer = millis();
      }
      break;

    case STATE_CAPPING:
      statusStr = "CAPPING";
      pumpState = false;
      servoPos = 90; // Servo Berputar 90 Derajat Memasang Tutup
      buzzerState = false;

      // Proses Capping Berlangsung selama 1.5 Detik
      if (millis() - stateTimer > 1500) {
        currentState = STATE_COMPLETED;
        bottleCount++; // Tambah Counter Botol
        stateTimer = millis();
      }
      break;

    case STATE_COMPLETED:
      statusStr = "DONE";
      pumpState = false;
      servoPos = 0; // Servo Kembali Naik
      buzzerState = true; // Sinyal Buzzer Selesai

      // Tahan status selesai & bunyi buzzer selama 1 Detik, lalu reset batch baru
      if (millis() - stateTimer > 1000) {
        currentState = STATE_FILLING;
      }
      break;
  }

  // Eksekusi Hardware
  digitalWrite(PUMP_RELAY_PIN, pumpState ? HIGH : LOW);
  capperServo.write(servoPos);
  digitalWrite(BUZZER_PIN, buzzerState ? HIGH : LOW);

  // 4. Kirim Telemetry Terformat ke Serial
  Serial.print("LEVEL:");
  Serial.print(actualVolume);
  Serial.print("|TARGET:");
  Serial.print(targetVolume);
  Serial.print("|PUMP:");
  Serial.print(pumpState ? 1 : 0);
  Serial.print("|SERVO:");
  Serial.print(servoPos);
  Serial.print("|COUNT:");
  Serial.print(bottleCount);
  Serial.print("|STATUS:");
  Serial.println(statusStr);

  delay(300);
}
