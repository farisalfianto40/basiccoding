int pinFlame = 5;
int pinAlarm = 2;
 
void setup() {
Serial.begin(9600);
pinMode(pinFlame, INPUT);
pinMode(pinAlarm, OUTPUT);
}
 
void loop() {
int statusApi = digitalRead(pinFlame);
digitalWrite(pinAlarm, statusApi);
if (statusApi == HIGH) {
Serial.println("API TERDETEKSI - Alarm ON");
} else {
Serial.println("AMAN - Alarm OFF");
}
delay(500);
}
