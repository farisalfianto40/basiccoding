Skenario Operasi High-Speed Quality Inspection Line

  1. Hubungkan komponen: Sensor Proximity IR ke Pin D2, Potensio Simulator Berat ke Pin A0, Servo Pusher ke Pin D3, Motor Getar ke Pin D5, dan Buzzer ke Pin D6.
  2. Putar Potensio 10K untuk mensimulasikan nilai bobot kemasan makanan ringan yang melintas (rentang 0 gram – 300 gram).
  3. Saat Sensor IR (D2) mendeteksi kemasan melintas di depan sensor, sistem secara instan mengukur bobot produk.
  4. Jika Bobot Standar (≥180g): Produk dinyatakan PASS. Servo tetap di posisi 0°, Motor Getar (D5) aktif menata kardus, dan kemasan lanjut ke dus packaging.
  5. Jika Bobot Kurang (<180g): Produk dinyatakan REJECT. Servo Motor Pusher (D3) berputar cepat 90° menendang produk ke Reject Bin, dan Buzzer berbunyi beep peringatan.
  6. Sistem secara otomatis memperbarui statistik Total Inspected, Passed, Rejected, serta Grafik Persentase secara real-time.

high_speed_rejector.ino 
// =========================================================================
// PROGRAM HIGH-SPEED QUALITY INSPECTION & REJECTOR (INDUSTRI PACKAGING/FMCG)
// MENGGUNAKAN SENSOR IR, POTENSIOMETER BERAT, SERVO PUSHER & MOTOR GETAR
// JURUSAN TEKNIK KIMIA & OTOMASI INDUSTRI - SMK SMTI YOGYAKARTA
// -------------------------------------------------------------------------
// Pin Digital 2 : Sensor Infrared Proximity (Active LOW when detected)
// Pin Analog A0 : Potensio 10K (Simulator Sensor Berat / Loadcell 0-300g)
// Pin Digital 3 : Servo Motor Pusher (Lengan Penendang Reject Bin)
// Pin Digital 5 : Motor Getar / Vibration Motor (Penata Dus Packaging)
// Pin Digital 6 : Buzzer Signal Defect / Reject Beep
// Baud Rate     : 9600 bps
// =========================================================================

#include <Servo.h>

#define IR_SENSOR_PIN 2
#define POT_WEIGHT_PIN A0
#define SERVO_PUSHER_PIN 3
#define VIBRO_MOTOR_PIN 5
#define BUZZER_PIN 6

Servo pusherServo;

const int MIN_WEIGHT_PASS = 180; // Standar Berat Minimum Produk Lolos (180 gram)

int totalInspected = 0;
int passedCount = 0;
int rejectedCount = 0;

bool lastIrState = HIGH;

void setup() {
  Serial.begin(9600);

  pinMode(IR_SENSOR_PIN, INPUT_PULLUP);
  pinMode(POT_WEIGHT_PIN, INPUT);
  pinMode(VIBRO_MOTOR_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  pusherServo.attach(SERVO_PUSHER_PIN);
  pusherServo.write(0); // Posisi Lengan Servo Standby (0 Derajat)

  digitalWrite(VIBRO_MOTOR_PIN, HIGH); // Motor Getar Penata Dus ON
  digitalWrite(BUZZER_PIN, LOW);
}

void loop() {
  // 1. Baca Sensor IR (Active LOW jika ada produk melintas)
  bool irDetected = (digitalRead(IR_SENSOR_PIN) == LOW);

  // 2. Baca Potensio Berat Produk (0 - 300 gram)
  int potRaw = analogRead(POT_WEIGHT_PIN);
  int weightGram = map(potRaw, 0, 1023, 0, 300);

  bool isReject = false;
  int servoPos = 0;
  String statusStr = "PASS";

  // Trigger Deteksi tepi saat produk baru masuk di depan Sensor IR
  if (irDetected && !lastIrState) {
    totalInspected++;

    if (weightGram < MIN_WEIGHT_PASS) {
      // PRODUK REJECT / KURANG BOBOT
      isReject = true;
      rejectedCount++;
      statusStr = "REJECT";
      servoPos = 90; // Pusher Mengayun Menendang Ke Reject Bin

      digitalWrite(BUZZER_PIN, HIGH);
      pusherServo.write(servoPos);
      delay(400); // Tahan Tendangan
      digitalWrite(BUZZER_PIN, LOW);
      pusherServo.write(0); // Kembali Standby
    } else {
      // PRODUK PASS / LOLOS STANDAR
      passedCount++;
      statusStr = "PASS";
      servoPos = 0;
    }
  }

  lastIrState = irDetected;

  // 3. Kirim Telemetry Terformat ke Serial
  Serial.print("IR:");
  Serial.print(irDetected ? 1 : 0);
  Serial.print("|WEIGHT:");
  Serial.print(weightGram);
  Serial.print("|SERVO:");
  Serial.print(servoPos);
  Serial.print("|VIBRO:");
  Serial.print(1);
  Serial.print("|TOTAL:");
  Serial.print(totalInspected);
  Serial.print("|PASSED:");
  Serial.print(passedCount);
  Serial.print("|REJECTED:");
  Serial.print(rejectedCount);
  Serial.print("|STATUS:");
  Serial.println(statusStr);

  delay(200);
}
