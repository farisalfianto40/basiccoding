Panduan Penggunaan

  1. Unggah sketch program Arduino ke modul Arduino Uno Anda.
  2. Hubungkan Flame Sensor ke Pin 2 dan Buzzer/Siren ke Pin 4.
  3. Gunakan browser Chrome atau Edge, lalu klik tombol "Hubungkan Arduino" di kanan atas.
  4. Pilih port USB COM Arduino Anda pada jendela pop-up.
  5. Uji sensor dengan korek api/sumber cahaya inframerah untuk memicu alarm fisik & animasi.

// ======================================================
// PROGRAM DETEKSI API GUDANG (SMK SMTI YOGYAKARTA)
// Pin Digital 2 : Flame Sensor (Digital Output)
// Pin Digital 4 : Buzzer / Alarm Siren
// Baud Rate     : 9600 bps
// ======================================================

const int FLAME_PIN  = 2;  // Pin masukan Flame Sensor
const int ALARM_PIN  = 4;  // Pin keluaran Buzzer/Sirine

void setup() {
  // Inisialisasi komunikasi Serial
  Serial.begin(9600);

  // Konfigurasi Pin Digital
  pinMode(FLAME_PIN, INPUT);
  pinMode(ALARM_PIN, OUTPUT);

  // Pasang sirine awal kondisi MATI
  digitalWrite(ALARM_PIN, LOW);
}

void loop() {
  // Membaca status sensor (Kebanyakan sensor flame aktif LOW / 0 saat ada api)
  int flameState = digitalRead(FLAME_PIN);

  if (flameState == LOW) {
    // API TERDETEKSI!
    digitalWrite(ALARM_PIN, HIGH);  // Hidupkan Buzzer/Sirine
    Serial.println("FIRE_DETECTED");  // Kirim data serial ke Web
  } else {
    // KONDISI AMAN
    digitalWrite(ALARM_PIN, LOW);   // Matikan Buzzer/Sirine
    Serial.println("SAFE");          // Kirim data serial ke Web
  }

  // Delay interaksi pembacaan (500 ms)
  delay(500);
}
