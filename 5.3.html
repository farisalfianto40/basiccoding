Simulasi Isolasi Bahan Bakar Tungku Peleburan (Reheating Furnace)

Pantau Sensor Flame (Digital) dan Sensor Suhu DHT22 secara realtime. Jika terdeteksi api liar atau suhu melampaui batas bahaya (>50°C), sistem memicu Emergency Shutdown: menutup Katup Gas Servo (0°), menyalakan Exhaust Fan 5V, dan mengaktifkan Sirine Alarm.

Panduan Penggunaan Simulasi

  1. Rangkai komponen di lab: Sensor Flame Digital (Pin D2), Sensor Suhu DHT22 (Pin D3), Relay Exhaust Fan 5V (Pin D4), Buzzer Alarm (Pin D5), dan Servo Motor Isolation Valve (Pin D9).
  2. Unggah sketch program Arduino C++ ke papan Arduino Uno.
  3. Klik tombol "Hubungkan Arduino" di atas dan pilih port COM USB yang sesuai.
  4. Uji kondisi normal: Katup gas terbuka 90°, exhaust fan mati, dan buzzer hening.
  5. Uji kondisi darurat: Berikan sumber api mendekati Sensor Flame ATAU naikkan suhu DHT22 melampaui batas bahaya (>50°C). Sistem akan memicu EMERGENCY SHUTDOWN: Servo memutar valve ke 0° untuk memutus pasokan gas, exhaust fan aktif menyedot asap, buzzer berbunyi nyaring, dan visualisator menampilkan efek kilat merah peringatan kebakaran.

// =========================================================================
// PROGRAM PROTEKSI KEBAKARAN & ISOLATION VALVE PIPA GAS (REHEATING FURNACE)
// JURUSAN TEKNIK METALURGI & MEKATRONIKA INDUSTRI - SMK SMTI YOGYAKARTA
// -------------------------------------------------------------------------
// Pin Digital 2 : Sensor Flame (Digital Input - LOW = Api Terdeteksi)
// Pin Digital 3 : Sensor Suhu DHT22
// Pin Digital 4 : Relay Kipas 5V (Exhaust Fan)
// Pin Digital 5 : Buzzer Alarm Kebakaran
// Pin Digital 9 : Servo Motor (Isolation Valve Pipa Gas)
// Baud Rate     : 9600 bps
// =========================================================================

#include <Servo.h>
#include <DHT.h>

#define FLAME_PIN 2
#define DHT_PIN 3
#define DHTTYPE DHT22
#define FAN_RELAY_PIN 4
#define BUZZER_PIN 5
#define SERVO_PIN 9

const float TEMP_THRESHOLD = 50.0; // Batas bahaya suhu (Derajat Celsius)

Servo gasValve;
DHT dht(DHT_PIN, DHTTYPE);

void setup() {
  Serial.begin(9600);

  pinMode(FLAME_PIN, INPUT);
  pinMode(FAN_RELAY_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  gasValve.attach(SERVO_PIN);
  dht.begin();

  // Condition awal normal
  gasValve.write(90); // Katup gas terbuka 90 derajat
  digitalWrite(FAN_RELAY_PIN, LOW);
  digitalWrite(BUZZER_PIN, LOW);
}

void loop() {
  // 1. Baca Sensor Flame (Modul digital LM393 aktif LOW saat mendeteksi api)
  int flameState = digitalRead(FLAME_PIN);
  bool isFlameDetected = (flameState == LOW);

  // 2. Baca Sensor Suhu DHT22
  float tempC = dht.readTemperature();
  if (isnan(tempC)) {
    tempC = 25.0; // Fallback jika sensor tidak terbaca
  }

  // 3. Evaluasi Kondisi Bahaya Kebakaran / Overheat
  bool isEmergency = isFlameDetected || (tempC >= TEMP_THRESHOLD);

  String statusStr = "NORMAL";
  int valveAngle = 90;
  int fanState = 0;
  int buzzerState = 0;

  if (isEmergency) {
    // KONDISI BAHAYA (EMERGENCY SHUTDOWN):
    statusStr = "EMERGENCY";
    valveAngle = 0;   // Mutus aliran gas cepat (0 derajat)
    fanState = 1;     // Nyalakan Exhaust Fan
    buzzerState = 1;  // Bunyikan Alarm

    gasValve.write(valveAngle);
    digitalWrite(FAN_RELAY_PIN, HIGH);
    digitalWrite(BUZZER_PIN, HIGH);
  } else {
    // KONDISI NORMAL:
    statusStr = "NORMAL";
    valveAngle = 90;  // Suplai gas terbuka normal (90 derajat)
    fanState = 0;     // Kipas Exhaust Mati
    buzzerState = 0;  // Alarm Mati

    gasValve.write(valveAngle);
    digitalWrite(FAN_RELAY_PIN, LOW);
    digitalWrite(BUZZER_PIN, LOW);
  }

  // 4. Kirim Data Serial Terformat ke Web Dashboard Visualizer
  Serial.print("FLAME:");
  Serial.print(isFlameDetected ? 1 : 0);
  Serial.print("|TEMP:");
  Serial.print(tempC, 1);
  Serial.print("|VALVE:");
  Serial.print(valveAngle);
  Serial.print("|FAN:");
  Serial.print(fanState);
  Serial.print("|BUZZER:");
  Serial.print(buzzerState);
  Serial.print("|STATUS:");
  Serial.println(statusStr);

  delay(500);
}
