Pengaman Pekerja Terkunci Dalam Cold Storage Beku

DHT22 memantau suhu minus (°C) gudang beku bahan pangan/obat. Solenoid Door Lock mengunci pintu. Jika Sensor PIR mendeteksi pekerja di dalam saat terkunci, sistem langsung mengaktifkan Unlock Override, Servo membuka ventilasi darurat, dan Siren/Buzzer berbunyi melengking.

Panduan Penggunaan Simulasi Safety Interlock

  1. Rangkai modul komponen: DHT22 Sensor (Pin D2), PIR Motion Sensor (Pin D3), Solenoid Lock 12V / Relay (Pin D4), Buzzer Siren (Pin D5), dan Servo Katup Ventilasi (Pin D9).
  2. Unggah program C++ ke mikrokontroler Arduino Uno.
  3. Klik tombol "Hubungkan Arduino" di atas dan pilih COM Port yang sesuai.
  4. Dalam kondisi normal tanpa gerak manusia, Solenoid mengunci pintu cold room (LOCKED) dan katup ventilasi tertutup.
  5. Lakukan gerakan di depan Sensor PIR. Sistem akan mendeteksi pekerja terjebak, secara otomatis melepas Solenoid (Unlock Override), membuka katup ventilasi darurat via Servo, dan membunyikan sirine alarm darurat.

// =========================================================================
// PROGRAM COLD STORAGE & TRAP DOOR INTERLOCK SAFETY SYSTEM
// INDUSTRI MAKANAN & FARMASI
// -------------------------------------------------------------------------
// Pin Digital 2 : DHT22 Sensor (Suhu & Kelembaban Cold Room)
// Pin Digital 3 : PIR Sensor (Deteksi Gerak Pekerja Terjebak)
// Pin Digital 4 : Solenoid Door Lock 12V (Relay Output)
// Pin Digital 5 : Buzzer Alarm Peringatan Melengking
// Pin Digital 9 : Servo Motor (Katup Sirkulasi Air Vent Darurat)
// Baud Rate     : 9600 bps
// =========================================================================

#include <DHT.h>
#include <Servo.h>

#define DHTPIN 2
#define DHTTYPE DHT22

const int PIR_PIN      = 3;
const int SOLENOID_PIN = 4;
const int BUZZER_PIN   = 5;
const int SERVO_PIN    = 9;

DHT dht(DHTPIN, DHTTYPE);
Servo ventServo;

void setup() {
  Serial.begin(9600);
  dht.begin();

  pinMode(PIR_PIN, INPUT);
  pinMode(SOLENOID_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  ventServo.attach(SERVO_PIN);
  ventServo.write(0); // Katup ventilasi tertutup rapat (0 derajat)
  
  digitalWrite(SOLENOID_PIN, HIGH); // Kondisi default: Pintu Terkunci
  digitalWrite(BUZZER_PIN, LOW);
}

void loop() {
  // 1. Pembacaan Suhu & Kelembaban Gudang Beku
  float temp = dht.readTemperature();
  float hum  = dht.readHumidity();

  // Fallback simulasi jika DHT22 belum terhubung hardware
  if (isnan(temp)) temp = -18.5;
  if (isnan(hum)) hum = 85.0;

  // 2. Pembacaan Status Sensor PIR Pekerja
  bool workerDetected = (digitalRead(PIR_PIN) == HIGH);

  bool solenoidLocked = true;
  int servoAngle = 0;
  bool buzzerActive = false;
  String statusStr = "LOCKED";

  // 3. LOGIKA SAFETY TRAP DOOR INTERLOCK:
  // Jika Pekerja Terdeteksi di dalam saat Cold Storage Beroperasi:
  if (workerDetected) {
    solenoidLocked = false; 
    digitalWrite(SOLENOID_PIN, LOW); // RELEASE / UNLOCK OVERRIDE PINTU

    servoAngle = 90; // Buka Katup Ventilasi Darurat
    ventServo.write(servoAngle);

    digitalWrite(BUZZER_PIN, HIGH); // Bunyikan Siren Alarm Melengking
    buzzerActive = true;

    statusStr = "TRAP_ALERT";
  } else {
    // Normal Operasional Cold Storage: Pintu Terkunci, Ventilasi Ditutup
    solenoidLocked = true;
    digitalWrite(SOLENOID_PIN, HIGH); // Solenoid Mengunci Pintu

    servoAngle = 0; // Ventilasi Tertutup
    ventServo.write(servoAngle);

    digitalWrite(BUZZER_PIN, LOW);
    buzzerActive = false;

    statusStr = "LOCKED";
  }

  // 4. Kirim Telemetry Serial Data ke Web Dashboard
  Serial.print("TEMP:");
  Serial.print(temp, 1);
  Serial.print("|HUM:");
  Serial.print(hum, 1);
  Serial.print("|PIR:");
  Serial.print(workerDetected ? 1 : 0);
  Serial.print("|LOCK:");
  Serial.print(solenoidLocked ? 1 : 0);
  Serial.print("|SERVO:");
  Serial.print(servoAngle);
  Serial.print("|BUZZER:");
  Serial.print(buzzerActive ? 1 : 0);
  Serial.print("|STATUS:");
  Serial.println(statusStr);

  delay(500);
}
