Menerima status string dari Arduino (contoh: RED_ON, YELLOW_ON, GREEN_ON).

Panduan Penggunaan

  1. Unggah sketch program Arduino ke modul Arduino Uno Anda.
  2. Hubungkan LED Merah ke Pin 8, LED Kuning ke Pin 9, dan LED Hijau ke Pin 10.
  3. Gunakan browser Chrome atau Edge, lalu klik tombol "Hubungkan Arduino" di kanan atas.
  4. Pilih port USB COM Arduino Anda pada jendela pop-up.
  5. Amati perubahan nyala LED yang berurutan secara otomatis setiap 2 detik.

lampu_berurutan.ino 
// ======================================================
// PROGRAM 3 LED BERURUTAN (MERAH, KUNING, HIJAU)
// Pin Digital 8 : LED Merah
// Pin Digital 9 : LED Kuning
// Pin Digital 10: LED Hijau
// Jeda Waktu    : 2 Detik (2000 ms)
// Baud Rate     : 9600 bps
// ======================================================

const int RED_PIN    = 8;   // Pin LED Merah
const int YELLOW_PIN = 9;   // Pin LED Kuning
const int GREEN_PIN  = 10;  // Pin LED Hijau

void setup() {
  // Inisialisasi komunikasi Serial
  Serial.begin(9600);

  // Konfigurasi Pin sebagai OUTPUT
  pinMode(RED_PIN, OUTPUT);
  pinMode(YELLOW_PIN, OUTPUT);
  pinMode(GREEN_PIN, OUTPUT);
}

void loop() {
  // 1. Nyalakan LED Merah, matikan yang lain
  digitalWrite(RED_PIN, HIGH);
  digitalWrite(YELLOW_PIN, LOW);
  digitalWrite(GREEN_PIN, LOW);
  Serial.println("RED_ON");
  delay(2000);  // Jeda 2 detik

  // 2. Nyalakan LED Kuning, matikan yang lain
  digitalWrite(RED_PIN, LOW);
  digitalWrite(YELLOW_PIN, HIGH);
  digitalWrite(GREEN_PIN, LOW);
  Serial.println("YELLOW_ON");
  delay(2000);  // Jeda 2 detik

  // 3. Nyalakan LED Hijau, matikan yang lain
  digitalWrite(RED_PIN, LOW);
  digitalWrite(YELLOW_PIN, LOW);
  digitalWrite(GREEN_PIN, HIGH);
  Serial.println("GREEN_ON");
  delay(2000);  // Jeda 2 detik
}
